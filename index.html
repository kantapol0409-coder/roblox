        speedLv = 9,
        houseLv = 10,
        houseSlots = "17/17",
        pets = {"🐲", "🦅", "🐺"},
        eggs = {"🟣", "🔥", "🪽"}
    }
    
    local json = HttpService:JSONEncode(payload)

            Url = "http://localhost:3000/api/update",
            Method = "POST",

            Body = json
        })
    end
end
